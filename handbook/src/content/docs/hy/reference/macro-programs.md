Մակրո ծրագիրը մակրո է, որը գրված է JavaScript-ով՝ քայլերի ցանկի փոխարեն։
Այն գործում է խմբագրիչի ներսում՝ փոքրիկ API-ի դեմ, որը կոչվում է `sound`, և թույլ է տալիս կարդալ
բաց նախագիծը, տեղափոխել ընտրությունը և կիրառել նույն ազդեցություններն ու հրամանները, որոնք կարող է կիրառել
քայլերի ցանկի մակրոն։ Այլ ամեն ինչ, սկսած ֆայլերից և ցանցից մինչև ձեր
այլ նախագծերը, գտնվում է դրա հասանելիության սահմաններից դուրս։

Ծրագրերը Soundscaper-ի հատկանիշն են։ Framescaper-ը չունի մակրո կառավարիչ։

## Ծրագրերի տեղակայումը

Ընտրեք **Գործիքներ → Մակրո կառավարիչ**։ Դիալոգային պատուհանը ցուցադրում է քայլերի ցանկի մակրոները և, **Ծրագրեր**
բաժնում, ձեր պահպանած ծրագրերը։ Սեղմեք **+ (Նոր ծրագիր)** կոճակը
Ծրագրերի վերնագրում՝ ստեղծելու մեկը։ Նույն գործողությունների վահանակը առաջարկում է **Ներմուծել ծրագիր**,
**Արտահանել ծրագիր** և **Ջնջել ծրագիր** ընտրված ծրագրի համար։
Մանրամասնության վահանակը ցուցադրում է դրա **Ծրագրի անունը**, **Ծրագրի** տեքստը և **Գործարկել
ծրագիր** կոճակը։ Տեքստը պահպանվում է ձեր գրելիս. առանձին պահպանման քայլ չկա։

Ծրագիրը պահվում է խմբագրիչի կարգավորումների հետ միասին, ոչ թե նախագծի ներսում, ուստի այն
հասանելի է յուրաքանչյուր նախագծում, որը բացում եք այս խմբագրիչում։ Օգտագործեք **Արտահանել ծրագիր** և
**Ներմուծել ծրագիր** գործողությունները՝ այն տեղափոխելու համար մեկ այլ մեքենա կամ մեկ այլ անձ. տե՛ս
[Ծրագրերի կիսումը](#sharing-programs)՝ դրա հետ կապված գործընթացների մասին։

[Ազդեցությունների նույն շղթայի կիրառումը յուրաքանչյուր անգամ](/guides/effects/apply-the-same-effects-every-time/)
մեկնաբանությունը ծածկում է նույն դիալոգային պատուհանի քայլերի ցանկի մասը։

## Ծրագրի գրում

Ծրագիրը `async` ֆունկցիայի մարմինն է, որը գործում է խիստ ռեժիմում։ Սա նշանակում է, որ
կարող եք `await` վերին մակարդակում, հայտարարել փոփոխականներ և ֆունկցիաներ, և օգտագործել բոլոր
սովորական լեզվական հատկանիշները։ `sound` օբյեկտը ծրագրի միակ կապն է
խմբագրիչի հետ, և դրա վրա կատարված յուրաքանչյուր կանչը վերադարձնում է խոստում (promise)։

```js
// Normalize everything, then fade the last two seconds.
await sound.select.all();
await sound.effect('audacity-normalize', { peakDb: -1 });
await sound.select.time(2, 0, { relativeTo: 'selection-end' });
await sound.effect('audacity-fade-out');
```

Tab ստեղծում է երկու բացատ ծրագրի դաշտում։ Escape սեղմեք, ապա Tab՝ դաշտից դուրս գալու համար։

### Ի՞նչ կարող է օգտագործել ծրագիրը

Սովորական JavaScript ստանդարտ գրադարանը ներկայացված է՝ `Object`, `Array`, `Map`,
`Set`, `Math`, `JSON`, `RegExp`, `Promise`, տիպավորված զանգվածները, `Intl`,
`TextEncoder`, `TextDecoder`, `structuredClone` և `queueMicrotask`։ `console`
նաև ներկայացված է, և ամեն ինչ, ինչ գրվում է դրա մեջ, հայտնվում է ծրագրի գրառման մեջ։

### Ի՞նչ չի կարող օգտագործել ծրագիրը

Ծրագիրը գործում է worker-ում, որի հնարավորությունները հանված են առաջին տողի գործարկումից առաջ։ Ծրագրի ներսում չկան հետևյալներից ոչ մեկը՝ `fetch`,
`XMLHttpRequest`, `WebSocket`, `indexedDB`, `caches`, `crypto`, `navigator`,
`location`, `Worker`, `WebAssembly`, `SharedArrayBuffer`, `Atomics`, `eval`,
`setTimeout` և `setInterval`։ Դրանցից ցանկացածը կարդալը տալիս է `undefined`։

Ծրագիրը չի կարող `import` մոդուլ։ Ստատիկ `import`-ը սինտակսիսի սխալ է այն տողում, որտեղ այն գտնվում է։ Ծրագրի ցանկացած անհրաժեշտությունը պետք է լինի ծրագրի ներսում։

Անվտանգության սահմանը բացակայող գլոբալ փոփոխականները չեն, այլ խմբագիրն ինքը՝ այն պատասխանում է միայն այս էջում նշված կանչերին և հրաժարվում է ամեն ինչից անունով, անկախ նրանից, թե ինչ է ծրագիրը կարողանում ուղարկել։

## Ծրագրի գործարկում

Սեղմեք **Գործարկել ծրագիր**։ Ամբողջ գործարկումը նախագծի պատմության մեկ գրառում է, ուստի մեկ **Հետ Վերցնել** գործողությունը հակադարձում է ամեն ինչ, ինչ արել է ծրագիրը, անկախ փոփոխությունների քանակից։
Եթե ծրագիրը սխալ է տալիս, ընդհատվում է կամ գերազանցում է իր ժամկետը, նախագիծը վերադարձվում է ճիշտ նույն վիճակում, ինչպես գործարկումից առաջ։

**Ընդհատել գործարկումը** անմիջապես կանգնեցնում է ծրագիրը։ Ծրագիրը, որը գործել է երկու րոպե, կանգնեցվում է նույն կերպ՝ *Մակրոն գործել է ավելի քան 120 վայրկյան։* հաղորդագրությամբ։

Գործարկումից հետո վահանակը ցուցադրում է ծրագրի գրառումը, իսկ գործարկումն ավարտվելու դեպքում՝ *Ծրագիրը կիրառվեց։*։
Չհաջողված գործարկումը ցուցադրում է *Ծրագիրը ձախողվեց N-րդ տողում՝* և սխալի հաղորդագրությունը, որտեղ տողի համարը ձեր ծրագրի այն տողն է, որը սխալ է տվել։

### Ի՞նչ ձայն է ազդեցությունը հպվում

Ծրագրի կողմից կիրառված ազդեցությունը գործում է ֆոկուսավորված հոսքի ընթացիկ ժամանակային ընտրության վրա, որը այն հոսքն է, որի վերնագիրը վերջին անգամ սեղմել եք կամ որի կլիպը վերջին անգամ ընտրել եք։ Եթե ժամանակային ընտրություն չկա, բայց կլիպ է ընտրված, ազդեցությունը ընդգրկում է այդ կլիպը։ Ծրագրի ընտրության կանչերը փոխում են ժամանակային տիրույթը և ընտրված հոսքերի հավաքածուն, բայց ոչ թե ֆոկուսավորված հոսքը, ուստի մեկ գործարկումը մշակում է մեկ հոսք։ Եթե ոչինչ չի ֆոկուսավորված կամ ընտրությունը դատարկ է, գործարկումը ձախողվում է նույն հաղորդագրությամբ, ինչը տալիս է Ազդեցությունների մենյուն։

## `sound` API

Ցանկացած մեթոդ ստորև վերադարձնում է promise, եթե այլ կերպ չի նշված։ Սպասեք յուրաքանչյուր կանչից հետո՝ հաջորդը կատարելուց առաջ. ծրագիրը, որը սկսում է ավելի քան ութ կանչ՝ առանց սպասելու, ութերորդը մերժվում է։

### `sound.env`

Պարզ առարկա, որը նկարագրում է գործարկումը։

| Դաշտ | Նշանակություն |
| --- | --- |
| `productId` | `"soundscaper"`։ |
| `locale` | Խմբագրի ինտերֆեյսի լեզուն, օրինակ՝ `"en"` կամ `"de"`։ |
| `seed` | Գործարկման պատահական թվերի սերմը։ Նոր է յուրաքանչյուր գործարկման համար։ |
| `startedAt` | Գործարկման սկսած իրական ժամանակը՝ որպես ISO 8601 շարք։ |
| `dryRun` | Ներկայումս միշտ `false`։ Պահուստ է։ |

### `sound.log`

`sound.log.info(...values)`, `sound.log.warn(...values)`,
`sound.log.error(...values)` և `sound.log.debug(...values)` գրում են մեկական տող գործարկման գրառման մեջ։ `console.log`, `console.info`, `console.warn`,
`console.error` և `console.debug` անում են նույնը։ Ոչ շարքային արժեքները գրվում են որպես JSON։ Այս մեթոդները ոչինչ չեն վերադարձնում և չեն պահանջում սպասել։

Գրառումը պարունակում է առավելագույնը 1,000 տող կամ 256 KiB, ինչը նախկինում է, և յուրաքանչյուր տողը կտրվում է 4,096 նիշից։ Այդ սահմանից ավելի տողերը ջնջվում և հաշվարկվում են. հաշվարկը հաղորդվում է որպես վերջնական զգուշացում։

### `sound.project`

Նախագծի կարդալը երբեք չի փոխում այն և չի հաշվարկվում գործարկման փոփոխությունների բյուջեի դեմ։

`sound.project.snapshot()` վերադարձնում է `{ sampleRate, tracks, selection }`, որտեղ
`tracks` և `selection`-ը վերադարձվում են ներքևի երկու կանչերի կողմից։ `sampleRate`-ը
նախագծի նմուշավորման հաճախությունն է հերցով, ինչը այն միավորն է, որով չաֳվում են այս էջի բոլոր կադրերի քանակները։

`sound.project.tracks()` վերադարձնում է հոսքերի զանգված՝ ժամանակագրության կարգով՝

```json
{ "id": "track-…", "name": "Voice", "kind": "audio", "index": 0, "muted": false, "solo": false }
```

`sound.project.clips(trackId)` վերադարձնում է մեկ հոսանքի կամ, եթե `trackId`-ը բաց թողնված է, բոլոր հոսանքների կլիպերը:

```json
{ "id": "clip-…", "name": "Take 1", "startFrame": 0, "durationFrames": 480000 }
```

`sound.project.selection()` վերադարձնում է ընթացիկ ընտրությունը:

```json
{ "startFrame": 0, "endFrame": 96000, "trackIds": ["track-…"] }
```

### `sound.select`

Յուրաքանչյուր ընտրության կանչը հաշվվում է որպես մեկ փոփոխություն և վերադարձնում է ստեղծած ընտրությունը՝ `sound.project.selection()`-ի վերադարձնած ձևով։

`sound.select.time(start, end, options)`-ը սահմանում է ժամանակային միջակայքը վայրկյաններով։ Այն Audacity-ի `SelectTime` հրամանն է, իսկ `options.relativeTo`-ը ընտրում է, թե որտեղից է չափվում յուրաքանչյուր եզրը։ Երկու եզրերն էլ կարող են լինել մինչև -100 վայրկյան։

| `relativeTo` | Սկզբնական եզր | Եզրավորող եզր |
| --- | --- | --- |
| `'project-start'` (լռելյայն) | `start` վայրկյան նախագծի սկզբից | `end` վայրկյան նախագծի սկզբից |
| `'project'` | `start` վայրկյան նախագծի սկզբից | `end` վայրկյան նախագծի ավարտից հետո |
| `'project-end'` | `start` վայրկյան նախագծի ավարտից առաջ | `end` վայրկյան նախագծի ավարտից առաջ |
| `'selection-start'` | `start` վայրկյան ընտրության սկզբից հետո | `end` վայրկյան ընտրության սկզբից հետո |
| `'selection'` | `start` վայրկյան ընտրության սկզբից հետո | `end` վայրկյան ընտրության ավարտից հետո |
| `'selection-end'` | `start` վայրկյան ընտրության ավարտից առաջ | `end` վայրկյան ընտրության ավարտից առաջ |

Նախագծի ավարտը վերջին կադրն է, որին հասնում է որևէ կլիպ։ Ընտրված հոսանքները մնում են իրենց նախկին վիճակում։

`sound.select.frames(startFrame, endFrame, options)`-ը սահմանում է ժամանակային միջակայքը կադրերով՝ նախագծի նմուշավորման հաճախականությամբ։ `options.trackIds`-ը անվանում է ընտրվող հոսանքները. երբ այն բաց է թողնվում, արդեն ընտրված հոսանքները մնում են ընտրված։ Միջակայքը սահմանափակվում է ժամանակագրության շրջանակներում, և եզրերը փոխվում են տեղերով, եթե հակադարձ են։

`sound.select.tracks(options)`-ը Audacity-ի `SelectTracks` հրամանն է։ Այն ընտրում է այն հոսանքները, որոնց ինդեքսը (հաշված 0-ից) գտնվում է `options.track`-ից (լռելյայն 0) `options.trackCount` հոսանքների ընդգրկման միջակայքում (լռելյայն 1)։ `options.mode`-ը `'set'` է՝ հոսանքների ընտրությունը փոխարինելու համար, `'add'`՝ լայնացնելու համար, կամ `'remove'`՝ այդ հոսանքները դրանից հանելու համար։ Ժամանակային միջակայքը մնում է իր նախկին վիճակում։

`sound.select.frequencies(options)`-ը Audacity-ի `SelectFrequencies` հրամանն է։ Այն սահմանում է սպեկտրալ ընտրությունը՝ `options.low` և `options.high` հերցով. բաց թողնված եզրը պահպանում է իր ներկայիս արժեքը։

`sound.select.all()`-ը ընտրում է ամբողջ նախագիծը բոլոր հոսանքներում։
`sound.select.none()`-ը մաքրում է ընտրությունը։

### `sound.effect(type, params)`

Կիրառում է մեկ էֆեկտ ընթացիկ ընտրության վրա՝ կենտրոնացված հոսանքի վրա։ `type`-ը [Էֆեկտներ, որոնք կարող է կիրառել ծրագիրը](#effects-a-program-can-apply)-ից էֆեկտի ID է,
իսկ `params`-ը այդ էֆեկտի պարամետրերի օբյեկտ է։ Բաց թողնված պարամետրերը վերցնում են էֆեկտի լռելյայն արժեքները. արժեքները ստուգվում են [Ձայնային էֆեկտների հղում](/reference/generated/audio-effects/)-ում նշված միջակայքերի համապատասխանությամբ։ Վերադարձնում է
`null`։

```js
await sound.effect('audacity-amplify', { gainDb: -3 });
```

### `sound.effects(steps)`

Կիրառում է էֆեկտների շղթա ընթացիկ ընտրության վրա մեկ անցքով, ճիշտ այնպես, ինչպես անելու էր քայլերի ցանկի մակրոն այդ քայլերով։ Յուրաքանչյուր քայլը `{ type, params }` է, և շղթան պահանջում է առնվազն մեկ քայլ։ Վերլուծվում է որպես `null`։

```js
await sound.effects([
  { type: 'audacity-remove-dc-offset' },
  { type: 'audacity-normalize', params: { peakDb: -3 } },
  { type: 'audacity-legacy-compressor', params: { thresholdDb: -18, ratio: 3 } },
]);
```

### `sound.command(name, params)`

Գործադրում է Audacity-ի մակրո հրամաններից մեկը, որը նշված է
[Հրամաններ, որոնք կարող է գործադրել ծրագիրը](#commands-a-program-can-run) բաժնում։ Չորս ընտրության
հրամանները ընդունում են այնտեղ նկարագրված պարամետրերը, իսկ մյուսները՝ ոչ մեկ։ Այնուհետև լուծում է
ընտրությունը։

```js
await sound.command('SelectTime', { start: 0, end: 5 });
await sound.command('Trim');
```

### `sound.runSaved(name)`

Գործարկում է քայլերի ցանկի մակրո, որը պահված է նույն մակրո մենեջերում, ըստ իր ճշգրիտ անվան, ներառյալ ցանկացած ընտրության հրամաններ, որոնք այն պարունակում է։ Պահպանված մակրոն ինքնուրույն չի կարող լինել ծրագիր, ուստի ծրագրերը չեն կարող նստվել։ Ապահովում է `null`-ը։ Անհայտ անունը մերժվում է։

### Ժամանակ և պատահականություն

Գործարկումը վերարտադրելի է. նույն ծրագրի երկու գործարկումները նույն նախագծի վրա կարդում են նույնը, քանի որ ժամացույցը և պատահական թվերը մեքենայինը չեն։

`Date.now()` և `new Date()`-ը առանց արգումենտների վերադարձնում են վիրտուալ ժամացույց, որը սկսվում է 0-ից և աճում է մեկով յուրաքանչյուր պատասխանված կանչի համար խմբագրիչը, և `ms`-ով յուրաքանչյուր `sound.wait(ms)`-ի համար։ `sound.wait`-ը անմիջապես լուծվում է. ծրագրի համար իրական ժամանակի համար կանգ առնելու ուղի չկա, և այն անհրաժեշտ չէ, քանի որ խմբագրիչին ուղղված յուրաքանչյուր կանչն ավարտվում է նրա խոստումից առաջ։

`Math.random()` և `sound.random()`-ը նույն գեներատորն են, սերիավորված `sound.env.seed`-ից։ Քննադատեք սերիան, եթե անհրաժեշտ է իմանալ, թե որ հաջորդականությունն է օգտագործվել գործարկման ժամանակ։

### Ձեր ենթադրությունների ստուգում

`sound.assert(condition, message)`-ը նետում է `message`-ը, երբ `condition`-ը կեղծ է։
`sound.assertEqual(actual, expected, message)`-ը համեմատում է երկու արժեքները որպես JSON և նետում է, երբ դրանք տարբեր են, հաղորդագրությամբ, որը անվանում է երկու արժեքները, եթե դուք չեք տալիս ոչինչ։ Քանի որ նետված սխալը ավարտում է գործարկումը և հետ շրջում ամեն ինչ նախքան այն, ձախողված ապացույցը նախագիծը անպատճառ է թողնում։ Երկու մեթոդներն էլ չեն վերադարձնում խոստում։

```js
const tracks = await sound.project.tracks();
sound.assert(tracks.length > 0, 'Import a recording first.');
await sound.command('SelectAll');
const selection = await sound.project.selection();
sound.assertEqual(selection.trackIds.length, tracks.length, 'Select all should cover every track.');
```

## Խանդեր, որոնք անցնում են խմբագրիչ

Ծրագրի փոխանցած յուրաքանչյուր արգումենտը և ստացած յուրաքանչյուր արժեքը պարզ տվյալներ են.
`null`, զրոյական արժեքներ, վերջավոր թվեր, շարքեր և այդ տվյալներից կազմված զանգվածներ և պարզ օբյեկտներ։ `NaN`, `Infinity`, ֆունկցիաներ, դասերի օրինակներ, տիպավոր զանգվածներ և `Date`
օբյեկտները մերժվում են սխալի հետ միասին, ինչպես նաև ցանկացած արժեք, որը ավելի մեծ է, քան 1 ՄԲ, ավելի խորը, քան 12 մակարդակ, կամ ունի ավելի քան 4,096 մուտք մեկ զանգվածում կամ օբյեկտում։
`undefined` հատկանիշները ջնջվում են։

## Սխալներ

Կանչը, որը խմբագրիչը մերժում է, մերժում է իր խոստումը `Error`-ով, որի `message`
բացատրում է, թե ինչու. հրաման, որը գտնվում է բառապաշարի սահմաններից դուրս, էֆեկտ դատարկ ընտրության վրա,
պարամետր, որը գտնվում է տիրույթի սահմաններից դուրս։ Սխալը նաև կրում է `code`, որը
`MACRO_CALL_FAILED` է, եթե խմբագրիչը չի տրամադրել ավելի կոնկրետ մեկը։ Ծրագիրը
կարող է բռնել դրանք և շարունակել։

```js
try {
  await sound.command('ExportWav');
} catch (error) {
  sound.log.warn(`refused: ${error.message}`);
}
```

Այդ ծրագիրը ավարտում է, և նրա օրագիրը կարդում է *refused: Unsupported macro command:
ExportWav.*

Սխալը, որը ծրագիրը չի բռնում, դադարեցնում է գործընթացը, վերադարձնում նախագիծը և ցուցադրվում է վահանակում՝ ներառյալ այն տողը, որտեղից այն եկել է։ Ծրագիրը, որը չի հավաքվում, նույն կերպ հաղորդվում է ցանկացած գործողությունից առաջ։

## Ծրագրի կողմից կատարվող հրամաններ {#commands-a-program-can-run}

`sound.command` ընդունում է ստորև նշված Audacity-ի մակրոհրամանների անվանումները։ Դրանք նույն անվանումներն են, որոնք կարող է պարունակել քայլերի ցանկի մակրոն, ուստի ծրագիրը և քայլերի ցանկը ունեն ճիշտ նույն հասանելիությունը։ Յուրաքանչյուր հրաման կատարում է խմբագրիչի գործողությունը, որը նկարագրված է [հրամանների հղումներում](/reference/generated/commands/)։

### Ընտրության հրամաններ պարամետրերով

Եթե թողնում եք պարամետրը դատարկ, ապա ընտրության այդ մասը մնում է անփոփոխ, ինչպես Audacity-ն էլ կարդում է դրանք։

### Ինչ է ինքնաբուխ բացակայում

`Undo` և `Redo` բացակայում են, քանի որ գործողությունը արդեն մեկ պատմության գրառում է, և քայլը, որը անցնում էր պատմության միջով, կհասներ գործողությանց ավելի և՛ ձեր սեփական խմբագրումներին։ Տրանսպորտային և գրանցման հրամանները բացակայում են, քանի որ ծրագիրը ոչինչ չի սպասում և չի կարող վերադարձվել գրանցումից։ Բացումը, պահպանումը, փակումը, ներմուծումը, արտահանումը և կարգավորումները բացակայում են, քանի որ ծրագրի հասանելիությունը սահմանափակված է նրանով, թե որ նախագիծն էր բացված, երբ այն սկսվեց։ Հրամանները, որոնք միայն բացում են երկխոսության պատուհան կամ փոխում են տեսքը, բացակայում են, քանի որ դրանք ոչինչ չեն փոխում նախագծում։

## Ծրագրերի կիսում {#sharing-programs}

**Ծրագրի արտահանումը** գրում է ընտրված ծրագիրը որպես `.soundscapemacro` ֆայլ, իսկ **Ծրագրի ներմուծումը** կարդում է այն։ Ֆայլը JSON է, ոչ թե մերկ `.js` ֆայլ, ուստի ընդունող համակարգիչը չի սխալի և չի մեկնաբանի այն որպես խմբագրիչից դուրս գործարկման համար նախատեսված բան։

```json
{
	"schemaVersion": 1,
	"kind": "script",
	"engine": "soundscaper-macro-js/1",
	"name": "Episode finish",
	"source": "await sound.select.all();\nawait sound.effect('audacity-normalize');\n"
}
```

Ներմուծումը պահպանում է միայն տեքստը և ոչինչ ավելի։ Ներմուծված ծրագիրը չունի **Ծրագիրը գործարկել** կոճակը։ Դրա փոխարեն՝ վահանակը ցույց է տալիս ծրագիրը, այն ֆայլը, որից այն եկել է, նշում այն մասին, թե ինչ կարող է անել ծրագիրը բաց նախագծի հետ, և փոքրիկ փակագիր՝ *Ես կարդացել եմ այս ծրագիրը և ցանկանում եմ այն գործարկել*։ Այն նշելիս ակտիվանում է **Ակտիվացնել այս ծրագիրը** կոճակը, և միայն այդ դեպքում կարող է ծրագիրը գործարկվել։

Այդ թույլտվությունը վերաբերում է ճիշտ նրան տեքստին, որը կարդացել եք։ Եթե ծրագիրը հետո փոխվի, ինչպես օրինակ՝ եթե դուք այն խմբագրեք կամ նոր տարբերակ ներմուծեք, ապա վերանայումը կրկին կհայտնվի, մինչև որ նոր տեքստը ակտիվացնեք։ Ծրագրերը, որոնք դուք ինքներդ գրում եք կառավարիչում, վերանայման կարիք չունեն։

## Օրինակներ

Կատարեք փոխանցում (fade in) առաջին հոսանքի բոլոր կլիպերի համար, որոնք ունեն փոխանցում։ Գործարկելուց առաջ սեղմեք այդ հոսանքի վերնագիրը, որպեսզի էֆեկտը հասնի այն հոսանքին, որը ծրագիրը կարդում է։

```js
let target = null;
let clips = [];
for (const track of await sound.project.tracks()) {
  clips = await sound.project.clips(track.id);
  if (clips.length) {
    target = track;
    break;
  }
}
sound.assert(target, 'There are no clips to fade.');
for (const clip of clips) {
  await sound.select.frames(clip.startFrame, clip.startFrame + clip.durationFrames, {
    trackIds: [target.id],
  });
  await sound.effect('audacity-fade-in');
  sound.log.info(`Faded in ${clip.name} on ${target.name}`);
}
```

Հաշվետվեք նախագծի մասին՝ այն փոխելու առանց:

```js
const { sampleRate, tracks } = await sound.project.snapshot();
for (const track of tracks) {
  const clips = await sound.project.clips(track.id);
  const frames = clips.reduce((total, clip) => total + clip.durationFrames, 0);
  sound.log.info(`${track.name}: ${clips.length} clip(s), ${(frames / sampleRate).toFixed(1)} s of audio`);
}
```

Գործարկեք պահպանված քայլերի զանգվածային հրամանը միայն այն դեպքում, երբ ընտրված տեքստը բավականաչափ երկար է։

```js
const { startFrame, endFrame } = await sound.project.selection();
const { sampleRate } = await sound.project.snapshot();
sound.assert((endFrame - startFrame) / sampleRate >= 30, 'Select at least thirty seconds.');
await sound.runSaved('Episode finish');
```

## Այս էջի մասին

Այս էջի վրա գտնվող յուրաքանչյուր ծրագիրը, մեկ տողանի փոքրիկ հատվածներից մինչև մանրամասն օրինակները,
փորձարկվում է Soundscaper-ի յուրաքանչյուր կառուցման համար՝ բրաուզերային փորձարկումների հավաքածուի կողմից
(`tests/browser/handbook-macro-program-examples.spec.js`), որը կարդում է ծրագրերը այս էջի սեփական տեքստից։ Ծրագիրը, որը դադարում է ավարտել կամ դադարում է արտադրել այն, ինչը այս էջը նշում է որպես արդյունք, ձախողում է կառուցումը, մինչև էջը կամ խմբագիրը ուղղվեն։
