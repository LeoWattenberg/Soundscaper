cmake_minimum_required(VERSION 3.28)
project(motionscaper_media_host VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
	set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG_901 REQUIRED IMPORTED_TARGET
	libavcodec=63.1.101 libavformat=63.1.101 libavutil=61.1.101 libswscale=10.1.101)
add_library(motionscaper-frame-carrier STATIC src/frame_carrier.cpp src/pcm_carrier.cpp
	../framescaper-media-host/src/strict_json.cpp ../common/sha256.cpp)
add_executable(motionscaper-media-host src/media_host.cpp src/prores_encoder.cpp)
target_link_libraries(motionscaper-media-host PRIVATE motionscaper-frame-carrier PkgConfig::FFMPEG_901)
if(MSVC)
	target_link_options(motionscaper-media-host PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/windows.manifest")
elseif(APPLE)
	target_link_options(motionscaper-media-host PRIVATE -Wl,-no_uuid)
elseif(NOT MSVC)
	target_link_options(motionscaper-media-host PRIVATE -Wl,--build-id=none)
endif()
include(CTest)
if(BUILD_TESTING)
	add_executable(motionscaper-frame-carrier-test tests/frame_carrier_fixture.cpp)
	target_link_libraries(motionscaper-frame-carrier-test PRIVATE motionscaper-frame-carrier)
	add_test(NAME motionscaper-frame-carrier COMMAND motionscaper-frame-carrier-test)
	add_executable(motionscaper-prores-roundtrip tests/prores_roundtrip.cpp src/prores_encoder.cpp)
	target_link_libraries(motionscaper-prores-roundtrip PRIVATE motionscaper-frame-carrier PkgConfig::FFMPEG_901)
	add_test(NAME motionscaper-codec-admission COMMAND motionscaper-media-host --self-test)
	add_test(NAME motionscaper-prores-roundtrip COMMAND motionscaper-prores-roundtrip "${CMAKE_BINARY_DIR}/roundtrip")
	add_executable(motionscaper-prores-edges-roundtrip tests/prores_edges_roundtrip.cpp src/prores_encoder.cpp)
	target_link_libraries(motionscaper-prores-edges-roundtrip PRIVATE motionscaper-frame-carrier PkgConfig::FFMPEG_901)
	add_test(NAME motionscaper-prores-edges-roundtrip COMMAND motionscaper-prores-edges-roundtrip "${CMAKE_BINARY_DIR}/edges-roundtrip")
	add_executable(motionscaper-pcm-roundtrip tests/pcm_roundtrip.cpp src/prores_encoder.cpp)
	target_link_libraries(motionscaper-pcm-roundtrip PRIVATE motionscaper-frame-carrier PkgConfig::FFMPEG_901)
	add_test(NAME motionscaper-pcm-roundtrip COMMAND motionscaper-pcm-roundtrip "${CMAKE_BINARY_DIR}/pcm-roundtrip")
endif()
set(MOTION_TARGETS motionscaper-frame-carrier motionscaper-media-host)
if(BUILD_TESTING)
	list(APPEND MOTION_TARGETS motionscaper-frame-carrier-test motionscaper-prores-roundtrip motionscaper-prores-edges-roundtrip motionscaper-pcm-roundtrip)
endif()
foreach(target IN LISTS MOTION_TARGETS)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /WX /permissive- /Brepro /utf-8)
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror
			-ffile-prefix-map=${CMAKE_SOURCE_DIR}=. -fdebug-prefix-map=${CMAKE_SOURCE_DIR}=. -fno-ident)
	endif()
endforeach()
install(TARGETS motionscaper-media-host RUNTIME DESTINATION bin)
