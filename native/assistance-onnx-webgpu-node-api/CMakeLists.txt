cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(soundscaper_onnx_webgpu_node_api LANGUAGES CXX)
find_package(Threads REQUIRED)
if(APPLE)
	set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version" FORCE)
endif()
if(MSVC)
	set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
	add_compile_options(/Brepro "/pathmap:${SCAPE_WORK_ROOT}=/scape/onnx-node-api" /EHsc)
	add_link_options(/Brepro)
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.def" "LIBRARY onnxruntime.dll\nEXPORTS\nOrtGetApiBase\n")
	if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
		set(ort_machine ARM64)
	else()
		set(ort_machine X64)
	endif()
	execute_process(COMMAND "${CMAKE_AR}" "/DEF:${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.def"
		"/OUT:${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.lib" "/MACHINE:${ort_machine}" /Brepro
		COMMAND_ERROR_IS_FATAL ANY)
	set(ORT_LINK_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.lib")
else()
	set(ORT_LINK_LIBRARY "${ORT_LIBRARY}")
	add_compile_options(-g0 "-ffile-prefix-map=${SCAPE_WORK_ROOT}=/scape/onnx-node-api"
		"-ffile-prefix-map=${ORT_HEADERS}=/scape/onnx-headers"
		"-fdebug-prefix-map=${SCAPE_WORK_ROOT}=/scape/onnx-node-api")
endif()
if(APPLE)
	set(CMAKE_INSTALL_RPATH "@loader_path")
else()
	set(CMAKE_INSTALL_RPATH "$ORIGIN/")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH FALSE)

file(GLOB ORT_NODE_SOURCES CONFIGURE_DEPENDS "${ORT_NODE_SOURCE}/*.cc")
add_library(onnxruntime_binding SHARED ${ORT_NODE_SOURCES} webgpu_plugin_provider.cc)
target_compile_features(onnxruntime_binding PRIVATE cxx_std_20)
target_compile_definitions(onnxruntime_binding PRIVATE
	NAPI_VERSION=6 ORT_API_MANUAL_INIT NODE_ADDON_API_CPP_EXCEPTIONS_ALL USE_WEBGPU=1)
if(APPLE)
	target_compile_definitions(onnxruntime_binding PRIVATE USE_COREML=1)
endif()
target_include_directories(onnxruntime_binding PRIVATE "${ORT_NODE_SOURCE}" "${ORT_HEADERS}"
	"${NODE_HEADERS}/include/node" "${NODE_ADDON_API}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(onnxruntime_binding PRIVATE "${ORT_LINK_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(onnxruntime_binding PROPERTIES PREFIX "" SUFFIX ".node")
if(WIN32)
	target_sources(onnxruntime_binding PRIVATE "${NODE_DELAY_HOOK}")
	target_compile_definitions(onnxruntime_binding PRIVATE HOST_BINARY="node.exe")
	target_link_libraries(onnxruntime_binding PRIVATE "${NODE_LIBRARY}" delayimp)
	target_link_options(onnxruntime_binding PRIVATE /DELAYLOAD:node.exe)
elseif(APPLE)
	target_link_options(onnxruntime_binding PRIVATE "-undefined" "dynamic_lookup")
endif()

add_executable(onnx_webgpu_probe webgpu_probe_main.cc)
target_compile_features(onnx_webgpu_probe PRIVATE cxx_std_20)
target_compile_definitions(onnx_webgpu_probe PRIVATE ORT_API_MANUAL_INIT)
target_include_directories(onnx_webgpu_probe PRIVATE "${ORT_HEADERS}")
target_link_libraries(onnx_webgpu_probe PRIVATE "${ORT_LINK_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(onnx_webgpu_probe PROPERTIES OUTPUT_NAME "onnx-webgpu-device-probe")

# Driver-independent regression: an earlier built-in EP with the same name must
# never replace the separately authenticated plugin. This fixture is not shipped.
add_executable(onnx_webgpu_device_test webgpu_device_test.cc)
target_compile_features(onnx_webgpu_device_test PRIVATE cxx_std_20)
target_compile_definitions(onnx_webgpu_device_test PRIVATE ORT_API_MANUAL_INIT)
target_include_directories(onnx_webgpu_device_test PRIVATE "${ORT_HEADERS}")
