cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(soundscaper_kokoro_g2p_gpu LANGUAGES CXX)
find_package(Threads REQUIRED)
foreach(required ORT_HEADERS ORT_LIBRARY PROBE_SOURCE SCAPE_WORK_ROOT)
  if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Missing authenticated ${required}")
  endif()
endforeach()
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version" FORCE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  add_compile_options(/Brepro "/pathmap:${SCAPE_WORK_ROOT}=/scape/g2p-gpu" /EHsc)
  add_link_options(/Brepro)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.def" "LIBRARY onnxruntime.dll\nEXPORTS\nOrtGetApiBase\n")
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    set(ort_machine ARM64)
  else()
    set(ort_machine X64)
  endif()
  execute_process(COMMAND "${CMAKE_AR}" "/DEF:${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.def"
    "/OUT:${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.lib" "/MACHINE:${ort_machine}" /Brepro COMMAND_ERROR_IS_FATAL ANY)
  set(ORT_LINK_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.lib")
else()
  add_compile_options(-g0 "-ffile-prefix-map=${SCAPE_WORK_ROOT}=/scape/g2p-gpu"
    "-ffile-prefix-map=${ORT_HEADERS}=/scape/onnx-headers")
  set(ORT_LINK_LIBRARY "${ORT_LIBRARY}")
endif()
add_library(scape_g2p_gpu SHARED ort-gemm.cc)
set_target_properties(scape_g2p_gpu PROPERTIES OUTPUT_NAME "scape-g2p-gpu")
add_executable(onnx_webgpu_probe "${PROBE_SOURCE}/webgpu_probe_main.cc")
set_target_properties(onnx_webgpu_probe PROPERTIES OUTPUT_NAME "onnx-webgpu-device-probe")
foreach(target scape_g2p_gpu onnx_webgpu_probe)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE ORT_API_MANUAL_INIT)
  target_include_directories(${target} PRIVATE "${ORT_HEADERS}" "${PROBE_SOURCE}")
  target_link_libraries(${target} PRIVATE "${ORT_LINK_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
endforeach()
